function(fiasco_add_example name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE fiasco)
  set_target_properties(${name} PROPERTIES FOLDER "examples")
endfunction()

fiasco_add_example(main main.cpp)
fiasco_add_example(hello basics/hello.cpp)
fiasco_add_example(routing_methods routing/methods.cpp)
fiasco_add_example(subrouters routing/subrouters.cpp)
fiasco_add_example(path_params params/path.cpp)
fiasco_add_example(query_params params/query.cpp)
fiasco_add_example(param_sources params/sources.cpp)
fiasco_add_example(body_model models/body.cpp)
fiasco_add_example(multiple_bodies models/multiple_bodies.cpp)
fiasco_add_example(collections models/collections.cpp)
fiasco_add_example(requiredness models/requiredness.cpp)
fiasco_add_example(constraints validation/constraints.cpp)
fiasco_add_example(dependencies dependencies.cpp)
fiasco_add_example(dependency_graph dependency_graph.cpp)
